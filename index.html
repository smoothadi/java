-- Procedure + Cursor
CREATE PROCEDURE ShowData()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE s_name VARCHAR(20);
    DECLARE s_marks INT;

    DECLARE c CURSOR FOR
        SELECT name,marks FROM Student;

    DECLARE CONTINUE HANDLER
    FOR NOT FOUND SET done=1;

    OPEN c;

    myloop: LOOP

        FETCH c INTO s_name,s_marks;

        IF done=1 THEN
            LEAVE myloop;
        END IF;

        SELECT CONCAT(
            s_name,' ',
            s_marks,' ',
            Bonus(s_marks)
        ) AS Result;

    END LOOP;

    CLOSE c;

END $$

DELIMITER ;

CALL ShowData();

CALL ShowData();

11. Forms, Menus and Reports


import mysql.connector

# Connect to MySQL
conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="root",
    database="college"
)

cursor = conn.cursor()

# Create Table
cursor.execute("""
CREATE TABLE IF NOT EXISTS students (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    marks INT
)
""")

# Insert Student
def insert_student():
    sid = int(input("Enter Student ID: "))
    name = input("Enter Student Name: ")
    marks = int(input("Enter Marks: "))

    sql = "INSERT INTO students VALUES (%s, %s, %s)"
    values = (sid, name, marks)

    cursor.execute(sql, values)
    conn.commit()

    print("Student Inserted Successfully\n")


# Get All Students
def get_students():
    cursor.execute("SELECT * FROM students")

    data = cursor.fetchall()

    print("\n--- Student List ---")

    for row in data:
        print("ID:", row[0], "| Name:", row[1], "| Marks:", row[2])

    print()


# Get Marks by Student ID
def get_marks():
    sid = int(input("Enter Student ID: "))

    sql = "SELECT name, marks FROM students WHERE id=%s"

    cursor.execute(sql, (sid,))

    data = cursor.fetchone()

    if data:
        print("\nName:", data[0])
        print("Marks:", data[1], "\n")
    else:
        print("Student Not Found\n")


# Menu
while True:

    print("===== STUDENT MANAGEMENT =====")
    print("1. Insert Student")
    print("2. Get All Students")
    print("3. Get Student Marks")
    print("4. Exit")

    choice = input("Enter Choice: ")

    if choice == '1':
        insert_student()

    elif choice == '2':
        get_students()

    elif choice == '3':
        get_marks()

    elif choice == '4':
        break

    else:
        print("Invalid Choice\n")

# Close Connection
conn.close()








12. Library Management SQL Query


CREATE TABLE Publisher (
    Pub_ID INT PRIMARY KEY,
    Pub_Name VARCHAR(50)
);

CREATE TABLE Book (
    Book_ID INT PRIMARY KEY,
    Title VARCHAR(50),
    Pub_ID INT,
    Author VARCHAR(50),
    FOREIGN KEY (Pub_ID)
    REFERENCES Publisher(Pub_ID)
);

CREATE TABLE Library_Branch (
    Branch_ID INT PRIMARY KEY,
    Branch_Name VARCHAR(50)
);

CREATE TABLE Book_Copies (
    Book_ID INT,
    Branch_ID INT,
    No_Of_Copies INT
);

SELECT
    B.Book_ID,
    B.Title,
    P.Pub_Name,
    B.Author,
    L.Branch_Name,
    BC.No_Of_Copies
FROM Book B
JOIN Publisher P
ON B.Pub_ID = P.Pub_ID
JOIN Book_Copies BC
ON B.Book_ID = BC.Book_ID
JOIN Library_Branch L
ON BC.Branch_ID = L.Branch_ID;
