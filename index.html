OOP · compile-time polymorphism · overloading
Concept: Method overloading allows multiple methods with the same name but different parameter lists. The compiler selects the appropriate method at compile time.

class MathOperation {

    int add(int a, int b) {
        return a + b;
    }

    int add(int a, int b, int c) {
        return a + b + c;
    }

    double add(double a, double b) {
        return a + b;
    }
}

    public static void main(String[] args) {
        MathOperation obj = new MathOperation();

        System.out.println("Sum of 2 doubles: "  + obj.add(5.5, 6.5));

import java.awt.*;
import java.awt.event.*;

public class Calculator1 extends Frame implements ActionListener {

    TextField T1;
    Button b1,b2,b3,b4,b5,b6,b7,b8,b9,b0,add,sub,mul,div,eq;
    int num1, num2, res;
    String op;

    Calculator1() {
        T1 = new TextField(20);
        add(T1);

        b0 = new Button("1");  b1 = new Button("2");
        b2 = new Button("3");  b3 = new Button("4");
        b4 = new Button("5");  b5 = new Button("6");
        b6 = new Button("7");  b7 = new Button("8");
        b8 = new Button("9");  b9 = new Button("0");

        add = new Button("+");  sub = new Button("-");
        mul = new Button("*");  div = new Button("/");
        eq  = new Button("=");

        // Add buttons to frame
        add(b1); add(b2); add(b3);
        add(b4); add(b5); add(b6);
        add(b7); add(b8); add(b9);
        add(b0); add(add); add(sub);
        add(mul); add(div); add(eq);

        // Register listeners
        b1.addActionListener(this);  b2.addActionListener(this);
        b3.addActionListener(this);  b4.addActionListener(this);
        b5.addActionListener(this);  b6.addActionListener(this);
        b7.addActionListener(this);  b8.addActionListener(this);
        b9.addActionListener(this);  b0.addActionListener(this);
        add.addActionListener(this); sub.addActionListener(this);
        mul.addActionListener(this); div.addActionListener(this);
        eq.addActionListener(this);

        setLayout(new FlowLayout());
        setVisible(true);
        setSize(250, 250);
    }

    public void actionPerformed(ActionEvent e) {
        String s = e.getActionCommand();

        if (s.equals("+") || s.equals("-") || s.equals("*") || s.equals("/")) {
            num1 = Integer.parseInt(T1.getText());
            op = s;
            T1.setText("");
        } else if (s.equals("=")) {
            num2 = Integer.parseInt(T1.getText());
            if (op.equals("+")) res = num1 + num2;
            if (op.equals("-")) res = num1 - num2;
            if (op.equals("*")) res = num1 * num2;
            if (op.equals("/")) res = num1 / num2;
            T1.setText(String.valueOf(res));
        } else {
            T1.setText(T1.getText() + s);
        }
    }

    public static void main(String args[]) {
        new Calculator1();
    }
}

Swing Event Handling
JFrame · JButton · ActionListener · background color
Concept: Demonstrates event-driven programming in Swing. Button clicks are detected via ActionListener and the frame background is changed dynamically.
EventHand.java Java 
import javax.swing.*;
import java.awt.*;
import java.awt.event.*;

public class EventHand extends JFrame implements ActionListener {

    JButton btnRed, btnBlue;

    public EventHand() {
        setTitle("Swing Event Observation");
        setLayout(new FlowLayout());
        setSize(300, 200);

        btnRed  = new JButton("Red");
        btnBlue = new JButton("Blue");

        btnRed.addActionListener(this);
        btnBlue.addActionListener(this);
        add(btnRed);
        add(btnBlue);

        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setVisible(true);
    }

    @Override
    public void actionPerformed(ActionEvent e) {
        if (e.getSource() == btnRed) {
            getContentPane().setBackground(Color.RED);
        } else if (e.getSource() == btnBlue) {
            getContentPane().setBackground(Color.BLUE);
        }
    }

    public static void main(String[] args) {
        new EventHand();
    }
}

Read & Write a File
java.io · BufferedWriter · FileWriter · character streams
Concept: Swing GUI application that writes user-entered text to a file using a BufferedWriter character stream. Appends data with each save operation.
FileObservation.java Java 
import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.io.*;

public class FileObservation extends JFrame implements ActionListener {

    JTextField textField;
    JButton    saveButton;

    public FileObservation() {
        setTitle("File Write Observation");
        setLayout(new FlowLayout());
        setSize(400, 150);

        add(new JLabel("Enter Text to Save:"));
        textField  = new JTextField(20);
        saveButton = new JButton("Save to File");

        add(textField);
        add(saveButton);
        saveButton.addActionListener(this);

        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setVisible(true);
    }

    @Override
    public void actionPerformed(ActionEvent e) {
        String content = textField.getText();
        try (BufferedWriter writer = new BufferedWriter(
                new FileWriter("observation.txt", true))) {
            writer.write(content);
            writer.newLine();
            JOptionPane.showMessageDialog(this, "Data Saved Successfully!");
            textField.setText("");
        } catch (IOException ex) {
            JOptionPane.showMessageDialog(this, "Error writing to file.");
        }
    }

    public static void main(String[] args) {
        new FileObservation();
    }
}

RMI Application
java.rmi · Remote · UnicastRemoteObject · Naming · Registry
Concept: Remote Method Invocation (RMI) allows a Java program to invoke methods on objects hosted on a different JVM. A remote interface, server implementation, registry, and client lookup are all combined here.
RMIStudentResult.java Java 
import java.net.MalformedURLException;
import java.rmi.*;
import java.rmi.server.*;
import java.rmi.registry.*;

// Remote Interface
interface ResultInterface extends Remote {
    String getResult(String name, int m1, int m2, int m3) throws RemoteException;
}

// Implementation
class ResultImpl extends UnicastRemoteObject implements ResultInterface {

    ResultImpl() throws RemoteException {}

    @Override
    public String getResult(String name, int m1, int m2, int m3) throws RemoteException {
        int    total = m1 + m2 + m3;
        double avg   = total / 3.0;
        return "Student Name: "   + name  +
               "\nTotal Marks: "   + total +
               "\nAverage Marks: " + avg;
    }
}

// Main Class — Server + Client
public class RMIStudentResult {
    public static void main(String args[]) {
        try {
            try {
                LocateRegistry.createRegistry(1099);
                System.out.println("RMI Registry created...");
            } catch (Exception e) {
                System.out.println("RMI Registry already exists.");
            }

            ResultImpl obj = new ResultImpl();
            Naming.rebind("rmi://localhost/ResultService", obj);
            System.out.println("Server Ready...");

            // Client lookup
            ResultInterface ref =
                (ResultInterface) Naming.lookup("rmi://localhost/ResultService");
            String result = ref.getResult("Karthi", 89, 96, 90);

            System.out.println("\nResult from RMI Server:");
            System.out.println(result);

        } catch (MalformedURLException | NotBoundException | RemoteException e) {
            e.printStackTrace();
        }
    }
}

ResultServlet.java Java 
import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;
import java.sql.*;

public class ResultServlet extends HttpServlet {

    public void doPost(HttpServletRequest req,
                       HttpServletResponse res)
                       throws IOException {

        res.setContentType("text/html");
        PrintWriter out = res.getWriter();

        String reg = req.getParameter("regno");

        try {
            Class.forName("com.mysql.cj.jdbc.Driver");

            Connection con = DriverManager.getConnection(
            "jdbc:mysql://localhost:3306/studentdb",
            "root","root");

            PreparedStatement ps =
            con.prepareStatement(
            "select * from result where regno=?");

            ps.setString(1, reg);

            ResultSet rs = ps.executeQuery();

            if(rs.next()) {
                out.println("Name: " +
                rs.getString("name"));

                out.println("<br>Status: " +
                rs.getString("status"));
            }
            else {
                out.println("No Record Found");
            }

            con.close();

        } catch(Exception e) {
            out.println(e);
        }
    }
}

Set Interface — Duplicate Email Finder

Concept: A HashSet automatically eliminates duplicates. retainAll() computes the intersection of two sets, effectively finding emails that appear in both lists.
DuplicateEmailFinder.java Java 
import java.util.*;

public class DuplicateEmailFinder {
    public static void main(String[] args) {

        List<String> list1 = Arrays.asList(
            "vishaal@gmail.com",
            "kirubak@gmail.com",
            "vishaal@gmail.com",
            "daniel9@gmail.com",
            "karthick@gmail.com",
            "kirubak@gmail.com"
        );

        List<String> list2 = Arrays.asList(
            "vishaal@gmail.com",
            "sangeeth@gmail.com",
            "vishnu@gmail.com",
            "kirubak@gmail.com"
        );

        Set<String> set1 = new HashSet<>(list1);
        Set<String> set2 = new HashSet<>(list2);
        set1.retainAll(set2);  // intersection

        System.out.println("Duplicate Emails:");
        for (String email : set1) {
            System.out.println(email);
        }
    }
}

FTP Using Sockets
java.net · ServerSocket · Socket · DataInputStream · DataOutputStream
Concept: Simulates file transfer over a TCP socket. The server listens on port 5000, accepts a connection, reads the filename/size, and writes the received bytes to disk. The client reads the file and streams it over the socket.
FTPServer.java Java 
import java.io.*;
import java.net.*;

public class FTPServer {
    public static void main(String[] args) {
        int port = 5000;
        try (ServerSocket serverSocket = new ServerSocket(port)) {
            System.out.println("Server started. Waiting for client...");
            Socket socket = serverSocket.accept();
            System.out.println("Client connected!");

            DataInputStream dis = new DataInputStream(socket.getInputStream());
            String fileName = dis.readUTF();
            long   fileSize = dis.readLong();

            FileOutputStream fos = new FileOutputStream("received_" + fileName);
            byte[] buffer    = new byte[4096];
            int    read;
            long   remaining = fileSize;

            while (remaining > 0 &&
                   (read = dis.read(buffer, 0, (int) Math.min(buffer.length, remaining))) != -1) {
                fos.write(buffer, 0, read);
                remaining -= read;
            }

            System.out.println("File received successfully!");
            fos.close(); dis.close(); socket.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
FTPClient.java Java 
import java.io.*;
import java.net.*;

public class FTPClient {
    public static void main(String[] args) {
        String server   = "localhost";
        int    port     = 5000;
        // Change path as needed
        String filePath =
            "C:\\Users\\Student\\Documents\\NetBeansProjects\\JavaApplication1\\src\\test.txt";

        try (Socket socket = new Socket(server, port)) {
            File             file = new File(filePath);
            FileInputStream  fis  = new FileInputStream(file);
            DataOutputStream dos  = new DataOutputStream(socket.getOutputStream());

            dos.writeUTF(file.getName());
            dos.writeLong(file.length());
            dos.flush();

            byte[] buffer = new byte[4096];
            int    read;
            while ((read = fis.read(buffer)) > 0) {
                dos.write(buffer, 0, read);
            }

            dos.flush();
            socket.shutdownOutput();
            System.out.println("File sent successfully!");
            fis.close(); dos.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
