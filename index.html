class Student {
    String name;
    int    rollNo;
    double marks;

    // 1. this to resolve field/parameter name conflict
    Student(String name, int rollNo, double marks) {
        this.name   = name;
        this.rollNo = rollNo;
        this.marks  = marks;
    }

    // 2. this() – constructor chaining
    Student(String name) {
        this(name, 0, 0.0);   // calls 3-arg constructor
    }

    // 3. this as method argument
    void register(Database db) { db.add(this); }

    // 4. this returned from a method (fluent builder)
    Student setName(String name)   { this.name = name;   return this; }
    Student setMarks(double marks) { this.marks = marks; return this; }

    void display() {
        System.out.printf("Name: %-15s Roll: %3d  Marks: %.1f%n",
                          name, rollNo, marks);
    }
}

class Database {
    void add(Student s) {
        System.out.println("Registered: " + s.name);
    }
}

package mypack;

public class MathUtils {
    public static int factorial(int n) {
        if (n <= 1) return 1;
        return n * factorial(n - 1);
    }
    public static boolean isPrime(int n) {
        if (n < 2) return false;
        for (int i = 2; i * i <= n; i++)
            if (n % i == 0) return false;
        return true;
    }
}

class InsufficientFundsException extends Exception {
    private double amount;
    InsufficientFundsException(double amount) {
        super("Insufficient funds! Short by: ₹" + amount);
        this.amount = amount;
    }
    double getShortfall() { return amount; }
}

class InvalidAgeException extends RuntimeException {
    InvalidAgeException(int age) {
        super("Invalid age: " + age + ". Must be 18–120.");
    }
}

// ── Bank class using checked exception ───────────────────────────
class BankAccount {
    private double balance;
    BankAccount(double initial) { balance = initial; }

    void withdraw(double amount) throws InsufficientFundsException {
        if (amount > balance)
            throw new InsufficientFundsException(amount - balance);
        balance -= amount;
        System.out.printf("Withdrew ₹%.2f | Balance: ₹%.2f%n", amount, balance);
    }
}

// ── Person class using unchecked exception ────────────────────────
class Person {
    void setAge(int age) {
        if (age < 18 || age > 120) throw new InvalidAgeException(age);
        System.out.println("Age set: " + age);
    }
}

public class ExceptionDemo {
    public static void main(String[] args) {
        // Checked exception
        BankAccount acc = new BankAccount(5000);
        try {
            acc.withdraw(2000);
            acc.withdraw(4000);  // will throw
        } catch (InsufficientFundsException e) {
            System.out.println("CAUGHT: " + e.getMessage());
            System.out.println("Shortfall: ₹" + e.getShortfall());
        } finally {
            System.out.println("Transaction attempt done.");
        }
        // Unchecked exception
        Person p = new Person();
        try {
            p.setAge(25);
            p.setAge(-5);
        } catch (InvalidAgeException e) {
            System.out.println("CAUGHT: " + e.getMessage());
        }
    }
}

import javax.swing.*;

public class Calculator extends JFrame implements ActionListener {

    private JTextField display = new JTextField("0");
    private double num1, num2, result;
    private char   op;
    private boolean start = true;

    private final String[][] KEYS = {
        {"C",  "±", "%", "÷"},
        {"7", "8", "9", "×"},
        {"4", "5", "6", "−"},
        {"1", "2", "3", "+"},
        {"0", ".", "=", "="}
    };

    Calculator() {
        setTitle("Calculator");
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLayout(new BorderLayout(5, 5));

        display.setFont(new Font("SansSerif", Font.BOLD, 28));
        display.setHorizontalAlignment(JTextField.RIGHT);
        display.setEditable(false);
        display.setBackground(Color.BLACK);
        display.setForeground(Color.WHITE);
        add(display, BorderLayout.NORTH);

        JPanel grid = new JPanel(new GridLayout(5, 4, 5, 5));
        for (String[] row : KEYS) {
            for (String label : row) {
                JButton btn = new JButton(label);
                btn.setFont(new Font("SansSerif", Font.BOLD, 18));
                btn.setBackground(isOp(label) ? new Color(255,149,0) : new Color(50,50,50));
                btn.setForeground(Color.WHITE);
                btn.addActionListener(this);
                grid.add(btn);
            }
        }
        add(grid, BorderLayout.CENTER);
        setSize(300, 400); setLocationRelativeTo(null); setVisible(true);
    }

    private boolean isOp(String s) {
        return s.equals("+")||s.equals("−")||s.equals("×")||s.equals("÷")||s.equals("=");
    }

    public void actionPerformed(ActionEvent e) {
        String cmd = e.getActionCommand();
        switch (cmd) {
            case "C": display.setText("0"); start = true; break;
            case "±": display.setText(String.valueOf(-Double.parseDouble(display.getText()))); break;
            case "%": display.setText(String.valueOf(Double.parseDouble(display.getText())/100)); break;
            case "=":
                num2 = Double.parseDouble(display.getText());
                switch (op) {
                    case '+': result = num1+num2; break;
                    case '−': result = num1-num2; break;
                    case '×': result = num1*num2; break;
                    case '÷': result = num2==0?Double.NaN:num1/num2; break;
                }
                display.setText(result == (long)result
                    ? String.valueOf((long)result)
                    : String.valueOf(result));
                start = true; break;
            default:
                if (isOp(cmd)) {
                    num1 = Double.parseDouble(display.getText());
                    op   = cmd.charAt(0); start = true;
                } else {
                    if (start) { display.setText(cmd); start = false; }
                    else       display.setText(display.getText() + cmd);
                }
        }
    }
    public static void main(String[] args) { new Calculator(); }
}

public class TempConverter extends JFrame {

    private JTextField inputField  = new JTextField(10);
    private JTextField outputField = new JTextField(10);
    private JLabel     resultLabel = new JLabel("  ---");
    private JRadioButton ftoc = new JRadioButton("°F → °C", true);
    private JRadioButton ctof = new JRadioButton("°C → °F");

    TempConverter() {
        setTitle("Temperature Converter");
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLayout(new GridBagLayout());
        GridBagConstraints g = new GridBagConstraints();
        g.insets = new Insets(8,8,8,8);
        g.fill = GridBagConstraints.HORIZONTAL;

        ButtonGroup bg = new ButtonGroup(); bg.add(ftoc); bg.add(ctof);
        JPanel rbPanel = new JPanel(); rbPanel.add(ftoc); rbPanel.add(ctof);

        JButton convertBtn = new JButton("Convert");
        JButton clearBtn   = new JButton("Clear");

        convertBtn.addActionListener(e -> {
            try {
                double val = Double.parseDouble(inputField.getText());
                double res = ftoc.isSelected()
                    ? (val - 32) * 5.0 / 9.0
                    : (val * 9.0 / 5.0) + 32;
                String unit = ftoc.isSelected() ? "°C" : "°F";
                outputField.setText(String.format("%.2f %s", res, unit));
                resultLabel.setText(String.format("%.2f → %.2f %s", val, res, unit));
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(this, "Enter a valid number!");
            }
        });
        clearBtn.addActionListener(e -> {
            inputField.setText(""); outputField.setText(""); resultLabel.setText("  ---");
        });

        // Layout
        addRow(g, new JLabel("Conversion:"), rbPanel,   0);
        addRow(g, new JLabel("Input:"),      inputField, 1);
        addRow(g, new JLabel("Result:"),     outputField,2);
        addRow(g, convertBtn, clearBtn, 3);

        pack(); setLocationRelativeTo(null); setVisible(true);
    }

    private void addRow(GridBagConstraints g, Component a, Component b, int row) {
        g.gridx=0; g.gridy=row; add(a, g);
        g.gridx=1;              add(b, g);
    }
    public static void main(String[] args) { new TempConverter(); }
}

// Extending Thread class
class DownloadThread extends Thread {
    private String filename;
    DownloadThread(String name, String file) {
        super(name); filename = file;
    }
    public void run() {
        for (int i = 10; i <= 100; i += 10) {
            System.out.printf("[%s] Downloading %s ... %d%%%n", getName(), filename, i);
            try { Thread.sleep(200); } catch (InterruptedException e) { e.printStackTrace(); }
        }
        System.out.println("[" + getName() + "] Download COMPLETE: " + filename);
    }
}

public class ThreadExtendDemo {
    public static void main(String[] args) throws InterruptedException {

        t1.setPriority(Thread.MAX_PRIORITY);
        t2.setPriority(Thread.MIN_PRIORITY);
        t1.start(); t2.start();
        t1.join(); t2.join();

    }
}

class Counter implements Runnable {
    private String name;
    private int    limit;
    Counter(String name, int limit) { this.name=name; this.limit=limit; }
    public void run() {
        for (int i = 1; i <= limit; i++) {
            System.out.printf("%-10s → %d%n", name, i);

        }
    }
}

// Producer-Consumer with synchronized block
class SharedBuffer {
    private int     data;
    private boolean available = false;

    public synchronized void produce(int val) throws InterruptedException {
        while (available) wait();
        data = val;
        available = true;
        System.out.println("Produced: " + val);
        notifyAll();
    }

    public synchronized int consume() throws InterruptedException {
        while (!available) wait();
        available = false;
        System.out.println("Consumed: " + data);
        notifyAll();
        return data;
    }
}

public class SyncDemo {
    public static void main(String[] args) {
        SharedBuffer buf = new SharedBuffer();
        new Thread(() -> {
            for (int i=1;i<=5;i++) try { buf.produce(i*10); } catch(Exception e){}
        }, "Producer").start();
        new Thread(() -> {
            for (int i=1;i<=5;i++) try { buf.consume();  } catch(Exception e){}
        }, "Consumer").start();
    }
}

import javax.swing.*;
import javax.swing.table.*;
import java.awt.*;
import java.util.*;

public class PersonalInfoSystem extends JFrame {

    private JTextField tfName, tfAge, tfEmail, tfPhone, tfCity, tfSearch;
    private DefaultTableModel model;
    private JTable table;

    PersonalInfoSystem() {
        setTitle("Personal Information System");
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLayout(new BorderLayout(10,10));
        ((JComponent)getContentPane()).setBorder(BorderFactory.createEmptyBorder(10,10,10,10));

        // --- FORM PANEL ---
        JPanel form = new JPanel(new GridLayout(6,2,8,8));
        form.setBorder(BorderFactory.createTitledBorder("Personal Details"));
        tfName  = addField(form, "Name:");
        tfAge   = addField(form, "Age:");
        tfEmail = addField(form, "Email:");
        tfPhone = addField(form, "Phone:");
        tfCity  = addField(form, "City:");
        tfSearch= addField(form, "Search:");
        add(form, BorderLayout.NORTH);

        // --- BUTTON PANEL ---
        JPanel btns = new JPanel(new FlowLayout());
        String[] bNames = {"Add","Update","Delete","Search","Clear"};
        for (String bn : bNames) {
            JButton b = new JButton(bn);
            b.addActionListener(e -> handle(bn));
            btns.add(b);
        }
        add(btns, BorderLayout.CENTER);

        // --- TABLE ---
        model = new DefaultTableModel(
            new String[]{"Name","Age","Email","Phone","City"}, 0);
        table = new JTable(model);
        table.getSelectionModel().addListSelectionListener(e -> fillForm());
        add(new JScrollPane(table), BorderLayout.SOUTH);

        setSize(700,550); setLocationRelativeTo(null); setVisible(true);
    }

    private JTextField addField(JPanel p, String label) {
        p.add(new JLabel(label)); JTextField tf = new JTextField(); p.add(tf); return tf;
    }

    private void handle(String action) {
        switch(action) {
            case "Add":
                model.addRow(new Object[]{tfName.getText(),tfAge.getText(),
                    tfEmail.getText(),tfPhone.getText(),tfCity.getText()});
                clearFields(); break;
            case "Delete":
                int row = table.getSelectedRow();
                if (row >= 0) model.removeRow(row); break;
            case "Update":
                int r = table.getSelectedRow();
                if (r >= 0) {
                    model.setValueAt(tfName.getText(),  r, 0);
                    model.setValueAt(tfAge.getText(),   r, 1);
                    model.setValueAt(tfEmail.getText(), r, 2);
                    model.setValueAt(tfPhone.getText(), r, 3);
                    model.setValueAt(tfCity.getText(),  r, 4);
                } break;
            case "Search":
                String q = tfSearch.getText().toLowerCase();
                for (int i=0;i<model.getRowCount();i++)
                    if(model.getValueAt(i,0).toString().toLowerCase().contains(q))
                      { table.setRowSelectionInterval(i,i); break; }
                break;
            case "Clear": clearFields(); break;
        }
    }

    private void fillForm() {
        int r = table.getSelectedRow();
        if (r < 0) return;
        tfName.setText(model.getValueAt(r,0).toString());
        tfAge.setText (model.getValueAt(r,1).toString());
        tfEmail.setText(model.getValueAt(r,2).toString());
        tfPhone.setText(model.getValueAt(r,3).toString());
        tfCity.setText (model.getValueAt(r,4).toString());
    }
    private void clearFields() {
        for (JTextField f : new JTextField[]{tfName,tfAge,tfEmail,tfPhone,tfCity,tfSearch})
            f.setText("");
    }
    public static void main(String[] args) { new PersonalInfoSystem(); }
}

        top.add(cb);

        // ActionListener — ComboBox
        combo = new JComboBox<>(new String[]{"Red","Green","Blue"});
        combo.addActionListener(e -> append("Combo: " + combo.getSelectedItem()));
        top.add(combo);

        // WindowListener
        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent e) { append("Window closing!"); }
        });

        // KeyListener

        keyField.setBorder(BorderFactory.createTitledBorder("Type here"));
        top.add(keyField);

        sliderLabel = new JLabel("Value: 50");

            sliderLabel.setText("Value: " + slider.getValue());
            append("Slider: " + slider.getValue());
        });

        canvas.setBackground(new Color(30,30,60));

    public void mouseClicked(MouseEvent e)  { append("Mouse CLICKED at ("+e.getX()+","+e.getY()+") button="+e.getButton()); }

    static void writeFile() throws IOException {
        String[] data = {
            "101,Alice,Computer Science,92.5",
            "102,Bob,Electronics,87.0",
            "103,Carol,Mechanical,95.0"
        };
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(FILE))) {
            bw.write("RollNo,Name,Dept,Marks\n");
            for (String line : data) { bw.write(line); bw.newLine(); }
        }
        System.out.println("File written: " + FILE);
    }

    // ── 2. Read using BufferedReader ───────────────────────────────
    static void readFile() throws IOException {
        System.out.println("\n─── File Contents ───");
        try (BufferedReader br = new BufferedReader(new FileReader(FILE))) {
            String line;
            while ((line = br.readLine()) != null) System.out.println(line);
        }
    }

    // ── 3. Append a record ────────────────────────────────────────
    static void appendFile() throws IOException {
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(FILE, true))) {
            bw.write("104,Dave,IT,88.5"); bw.newLine();
        }
        System.out.println("\nRecord appended.");
    }

    // ── 4. Search in file ─────────────────────────────────────────
    static void searchFile(String keyword) throws IOException {
        System.out.println("\n─── Search: '" + keyword + "' ───");
        try (BufferedReader br = new BufferedReader(new FileReader(FILE))) {
            br.lines()
              .filter(l -> l.toLowerCase().contains(keyword.toLowerCase()))
              .forEach(System.out::println);
        }
    }

    // ── 5. Count lines / words ────────────────────────────────────
    static void countFile() throws IOException {
        long lines = Files.lines(Paths.get(FILE)).count();
        long words = Files.lines(Paths.get(FILE))
                          .flatMap(l -> Arrays.stream(l.split(","))).count();
        System.out.printf("%nLines: %d  |  Fields: %d%n", lines, words);
    }

    public static void main(String[] args) throws IOException {
        writeFile();
        readFile();
        appendFile();
        readFile();
        searchFile("alice");
        countFile();
    }

public class MathServiceImpl extends UnicastRemoteObject implements MathService {

    MathServiceImpl() throws RemoteException { super(); }

    public int    add(int a, int b)      { return a + b; }
    public int    subtract(int a, int b) { return a - b; }
    public int    multiply(int a, int b) { return a * b; }
    public double divide(int a, int b)   { return b==0?Double.NaN:(double)a/b; }
    public long   factorial(int n) {
        long f=1; for(int i=2;i<=n;i++) f*=i; return f;
    }

    public static void main(String[] args) throws Exception {
        MathServiceImpl obj = new MathServiceImpl();
        Registry reg = LocateRegistry.createRegistry(1099);
        reg.rebind("MathService", obj);
        System.out.println("RMI Server ready on port 1099...");
    }

        this.salary=sal; this.email=email;
    }

    public String  getName()       { return name; }
    public String  getDepartment() { return department; }
    public double getSalary()     { return salary; }
    public String  getEmail()      { return email; }

    // Setters with validation
    public void setEmpId(int id)          { this.empId = id; }
    public void setName(String n)          { this.name = n; }
    public void setDepartment(String d)    { this.department = d; }
    public void setSalary(double s)        {
        if (s < 0) throw new IllegalArgumentException("Salary cannot be negative");
        this.salary = s;
    }
    public void setEmail(String em) {
        if (!em.contains("@")) throw new IllegalArgumentException("Invalid email");
        this.email = em;
    }

    public String toString() {
        return String.format("EmpID:%-5d | %-12s | %-15s | Salary: %8.2f | %s",
                             empId, name, department, salary, email);
    }
}

import java.util.*;

public class PhoneBook {

    private HashMap<String, List<String>> contacts = new HashMap<>();

    // Add a contact (supports multiple numbers per name)
    void addContact(String name, String phone) {
        contacts.computeIfAbsent(name, k -> new ArrayList<>()).add(phone);
        System.out.println("Added: " + name + " → " + phone);
    }

    // Search by exact name
    void search(String name) {
        List<String> nums = contacts.get(name);
        if (nums != null)
            System.out.println("📞 " + name + " → " + nums);
        else
            System.out.println("Contact '" + name + "' not found.");
    }

    // Partial-match search
    void partialSearch(String query) {
        System.out.println("\n─── Partial Search: '" + query + "' ───");
        contacts.entrySet().stream()
            .filter(e -> e.getKey().toLowerCase().contains(query.toLowerCase()))
            .forEach(e -> System.out.println(e.getKey() + " → " + e.getValue()));
    }

    // Delete a contact
    void delete(String name) {
        if (contacts.remove(name) != null)
            System.out.println("Deleted: " + name);
        else
            System.out.println("Not found: " + name);
    }

    // Display all contacts sorted
    void displayAll() {
        System.out.println("\n─── All Contacts (" + contacts.size() + ") ───");
        new TreeMap<>(contacts).forEach((k,v) ->
            System.out.printf("%-20s %s%n", k, v));
    }

    public static void main(String[] args) {
        PhoneBook pb = new PhoneBook();
        pb.addContact("Alice Johnson",  "9876543210");
        pb.addContact("Bob Smith",      "9123456789");
        pb.addContact("Carol White",    "9000011111");
        pb.addContact("Alice Johnson",  "9876543999"); // second number
        pb.addContact("David Brown",    "8888888888");

        System.out.println();
        pb.search("Bob Smith");
        pb.search("Unknown");
        pb.partialSearch("alice");

        pb.delete("Carol White");
        pb.displayAll();
    }
}

Find Duplicate Emails using Set Interface

DuplicateEmailFinder.java
import java.util.*;
import java.util.stream.*;

public class DuplicateEmailFinder {

    // Method 1: Using HashSet to detect duplicates
    static Set<String> findDuplicates(List<String> emails) {
        Set<String> seen  = new HashSet<>();
        Set<String> dupes = new LinkedHashSet<>(); // preserves insertion order
        for (String email : emails)
            if (!seen.add(email.toLowerCase()))
                dupes.add(email.toLowerCase());
        return dupes;
    }

    // Method 2: Using streams + groupingBy (frequency count)
    static Map<String,Long> frequencyMap(List<String> emails) {
        return emails.stream()
            .map(String::toLowerCase)
            .collect(Collectors.groupingBy(e -> e, Collectors.counting()));
    }

    // Method 3: Unique emails only (TreeSet = sorted)
    static Set<String> uniqueEmails(List<String> emails) {
        return emails.stream()
            .map(String::toLowerCase)
            .collect(Collectors.toCollection(TreeSet::new));
    }

    public static void main(String[] args) {

        List<String> emails = Arrays.asList(
            "alice@kit.edu", "bob@gmail.com",  "ALICE@KIT.EDU",
            "carol@yahoo.com","bob@gmail.com",  "dave@hotmail.com",
            "carol@yahoo.com","eve@outlook.com", "alice@kit.edu"
        );

        System.out.println("Input  (" + emails.size() + ") : " + emails);

        Set<String> dupes = findDuplicates(emails);
        System.out.println("\nDuplicates (" + dupes.size() + "):");
        dupes.forEach(e -> System.out.println("  ⚠  " + e));

        System.out.println("\nFrequency Map:");
        frequencyMap(emails).entrySet().stream()
            .sorted(Map.Entry.comparingByKey())
            .forEach(en -> System.out.printf("  %-25s × %d%n", en.getKey(), en.getValue()));

        System.out.println("\nUnique Emails (sorted):");
        uniqueEmails(emails).forEach(e -> System.out.println("  ✓ " + e));

        System.out.printf("%nTotal: %d → Unique: %d → Duplicates: %d%n",
            emails.size(), uniqueEmails(emails).size(), dupes.size());
    }
}

FTPServer.java
import java.io.*;
import java.net.*;

public class FTPServer {

    static final int PORT = 5000;
    static final String SERVER_DIR = "server_files/";

    public static void main(String[] args) throws IOException {
        new File(SERVER_DIR).mkdirs();
        ServerSocket ss = new ServerSocket(PORT);
        System.out.println("FTP Server started on port " + PORT);

        while (true) {
            Socket client = ss.accept();
            System.out.println("Client connected: " + client.getInetAddress());
            new Thread(() -> handleClient(client)).start();
        }
    }

    static void handleClient(Socket client) {
        try (
            DataInputStream  dis = new DataInputStream(client.getInputStream());
            DataOutputStream dos = new DataOutputStream(client.getOutputStream())
        ) {
            while (true) {
                String cmd = dis.readUTF();
                System.out.println("CMD: " + cmd);

                if (cmd.equalsIgnoreCase("QUIT")) { dos.writeUTF("Bye!"); break; }

                else if (cmd.startsWith("PUT")) {         // Upload
                    String fname = cmd.substring(4).trim();
                    long   size  = dis.readLong();
                    byte[] buf   = new byte[4096];
                    try (FileOutputStream fos = new FileOutputStream(SERVER_DIR+fname)) {
                        long remaining = size;
                        while (remaining > 0) {
                            int read = dis.read(buf, 0, (int)Math.min(buf.length, remaining));
                            fos.write(buf, 0, read); remaining -= read;
                        }
                    }
                    dos.writeUTF("PUT_OK: " + fname + " (" + size + " bytes)");
                }

                else if (cmd.startsWith("GET")) {         // Download
                    String fname = cmd.substring(4).trim();
                    File   f = new File(SERVER_DIR + fname);
                    if (!f.exists()) { dos.writeUTF("ERR: File not found"); continue; }
                    dos.writeUTF("GET_OK");
                    dos.writeLong(f.length());
                    try (FileInputStream fis = new FileInputStream(f)) {
                        byte[] buf = new byte[4096]; int n;
                        while ((n = fis.read(buf)) != -1) dos.write(buf, 0, n);
                    }
                }

                else if (cmd.equalsIgnoreCase("LIST")) {    // List files
                    File[] files = new File(SERVER_DIR).listFiles();
                    StringBuilder sb = new StringBuilder();
                    if (files != null)
                        for (File fi : files) sb.append(fi.getName()).append("  ").append(fi.length()).append("B\n");
                    dos.writeUTF(sb.length()==0?"(empty)":sb.toString());
                }

                else dos.writeUTF("ERR: Unknown command");
            }
        } catch (Exception e) { e.printStackTrace(); }
    }
}

        dos.writeLong(f.length());
        try (FileInputStream fis = new FileInputStream(f)) {
            byte[] buf = new byte[4096]; int n;
            while ((n = fis.read(buf)) != -1) dos.write(buf, 0, n);
        }
        System.out.println("Upload: " + dis.readUTF());
    }

                int read = dis.read(buf, 0, (int)Math.min(buf.length, remaining));
                fos.write(buf, 0, read); remaining -= read;
            }
        }
        System.out.println("Download complete: " + local + " (" + size + " bytes)");
    }
